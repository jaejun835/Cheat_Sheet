# 16_공격후정리

ACL 변경은 영구적으로 남는다. Event ID 5136으로 변경이 기록되므로 작업 후 원상복구를 해야 한다.

```bash
# ACL 제거
impacket-dacledit <도메인>/<공격자계정>:'<패스워드>' \
  -dc-ip <DC_IP> \
  -action remove \
  -rights DCSync \
  -principal <공격자계정> \
  -target-dn 'DC=<도메인>,DC=LOCAL'  # ← 수정

# 그룹 멤버 제거
bloodyAD --host <DC_IP> -d <도메인> -u <공격자계정> -p '<패스워드>' \
  remove groupMember '<대상그룹>' <공격자계정>  # ← 수정

# SPN 제거
bloodyAD --host <DC_IP> -d <도메인> -u <공격자계정> -p '<패스워드>' \
  set object <대상유저> servicePrincipalName -v ''  # ← 수정

# Shadow Credentials 제거
bloodyAD --host <DC_IP> -d <도메인> -u <공격자계정> -p '<패스워드>' \
  remove shadowCredentials <대상>  # ← 수정
```

```powershell
Remove-DomainObjectAcl -TargetIdentity "DC=<도메인>,DC=LOCAL" `
  -PrincipalIdentity <공격자계정> -Rights DCSync -Verbose  # ← 수정
Remove-DomainGroupMember -Identity '<대상그룹>' -Members <공격자계정> -Verbose
```
