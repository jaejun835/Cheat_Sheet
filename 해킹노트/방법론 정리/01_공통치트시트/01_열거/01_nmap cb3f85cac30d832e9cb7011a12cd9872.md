# 01_nmap

## 기본 스캔 순서 (표준 플로우)

### Tcp 스캔

```bash
# 1단계: 내부망 대역 확인 

ip route
cat /etc/hosts

###----------------------------------------------------------###

# 2단계: 호스트 스캔 (-sn = ping sweep)

nmap -sn --min-rate 2000 10.10.10.0/24 \
  | awk '/Nmap scan report/{print $NF}' \
  | tr -d '()'
  
# ping sweep에서 안 잡힐 때 TCP 기반 확인

nmap -Pn -p22,80,443,445,3389 --open 10.10.10.0/24

###----------------------------------------------------------###

# 3단계: 빠른 전체 TCP 포트 스캔

nmap -p- -Pn --min-rate 3000 <IP> -oA ports

###----------------------------------------------------------###

# 4단계: 열린 포트 추출

grep -oP '\d+(?=/open/tcp)' ports.gnmap | paste -sd, -

###----------------------------------------------------------###

# 5단계: 열린 포트 상세 스캔

nmap -p<열린포트들> -Pn -A <IP> -oA detail

###----------------------------------------------------------###

# 도메인 이름이 실제 확인된 경우

echo "<IP> <DOMAIN>" | sudo tee -a /etc/hosts
```

### UDP 스캔

```bash
# 1단계: 내부망 대역 확인 - UDP

ip route
cat /etc/hosts

###----------------------------------------------------------###

# 2단계: 호스트 스캔

nmap -sn --min-rate 2000 10.10.10.0/24 \
  | grep "report for" \
  | awk '{print $5}'
  
###----------------------------------------------------------###

# 3단계: 전체 UDP 포트 스캔

sudo nmap -sU -p- -Pn <IP> -oA udp_ports.txt

###----------------------------------------------------------###

# 4단계: open / open|filtered UDP 포트만 분리하기 (1번 아니면 2번 사용)

# 1번: 일반 출력에서 추출
cat udp_ports.txt.nmap \
  | grep -E 'open|open\|filtered' \
  | cut -d '/' -f 1 \
  | tr '\n' ','

# 2번: grepable 출력에서 추출
grep open udp_ports.txt.gnmap \
  | grep -oP '\d+/(open|open\|filtered)/udp' \
  | cut -d'/' -f1 \
  | tr '\n' ','
  
###----------------------------------------------------------###

# 5단계: 후보 UDP 포트만 상세 스캔

sudo nmap -sU -sV -Pn \
  --version-all \
  -p<UDP포트들> <IP> \
  -oA udp_detail.txt
  
###----------------------------------------------------------###

# 6단계: open|filtered 또는 unknown 포트만 다시 정밀 확인

sudo nmap -sU -sV -Pn \
  --version-all \
  --version-trace \
  -p<의심포트들> <IP> \
  -oA udp_retry.txt

###----------------------------------------------------------###

# 도메인 이름 나올 경우

echo "<IP> htb.local" | sudo tee -a /etc/hosts
```

## 스캔 타입

```bash
# SYN 스캔 (은밀한 스캔)

nmap -sS <IP>

###----------------------------------------------------------###

# TCP Connect 스캔 

nmap -sT <IP>

###----------------------------------------------------------###

# UDP 스캔

nmap -sU <IP>
nmap -sU --top-ports 20 <IP>
nmap -sU -p 53,67,68,69,123,161,162,500,514,1194 <IP>

###----------------------------------------------------------###

# TCP + UDP 동시

nmap -sS -sU -p T:80,443,U:53,161 <IP>

###----------------------------------------------------------###

# ACK 스캔 (방화벽 규칙 파악, 빠른 열린 포트 스캔 가능) 

nmap -sA <IP>

###----------------------------------------------------------###

# Null/FIN/Xmas (IDS 우회)

nmap -sN <IP>    # Null
nmap -sF <IP>    # FIN
nmap -sX <IP>    # Xmas

###----------------------------------------------------------###

# Ping 스캔 (호스트 발견)

nmap -sn <네트워크>/24
```

## 서비스 / 버전 / OS 탐지

```bash
# 서비스 버전 탐지

nmap -sV <IP>
nmap -sV --version-intensity 9 <IP>    # 최고 강도

###----------------------------------------------------------###

# OS 탐지

nmap -O <IP>
nmap -O --osscan-guess <IP>            # 추측 포함

###----------------------------------------------------------###

# 기본 스크립트 + 버전 (가장 자주 사용)

nmap -sC -sV <IP>
nmap -sC -sV -O <IP>                   # OS 포함
nmap -A <IP>                           # -sC -sV -O -traceroute 합산
```

## NSE 스크립트

```bash
# Nmap NSE 주요 카테고리
# auth       : 인증 방식, 익명 접근, 계정 관련 설정 검사
# broadcast  : 로컬 네트워크 브로드캐스트 기반 탐색
# brute      : 로그인 브루트포스. 계정 잠금과 IP 차단 주의
# default    : -sC로 실행되는 기본 스크립트 모음
# discovery  : 호스트명, 공유, 사용자, 서비스와 시스템 정보 열거
# dos        : 서비스 중단 가능성이 있어 일반 검사에 사용하지 않음
# exploit    : 취약점을 실제로 트리거할 수 있는 공격적 스크립트
# external   : 외부 API나 인터넷 서비스로 정보를 조회
# fuzzer     : 비정상 입력을 반복 전송하며 서비스 불안정 가능
# intrusive  : 대상에 영향을 줄 수 있는 침입성 검사
# malware    : 악성코드나 백도어 흔적 검사
# safe       : 상대적으로 영향이 적은 정보 확인 중심
# version    : 서비스 버전 탐지 보조
# vuln       : 알려진 취약점 후보 또는 취약 동작 검사

###----------------------------------------------------------###

# 기본 NSE 검사
# -sC는 --script=default와 동일

sudo nmap -Pn -sC -sV \
  -p<PORTS> <IP> \
  -oA 01_default

# default 카테고리를 명시적으로 실행

sudo nmap -Pn -sV \
  --script=default \
  -p<PORTS> <IP> \
  -oA 01_default

###----------------------------------------------------------###

# 상대적으로 영향이 적은 정보 확인

sudo nmap -Pn -sV \
  --script=safe \
  -p<PORTS> <IP> \
  -oA 02_safe

###----------------------------------------------------------###

# 호스트명, 공유, 사용자, 서비스 정보 추가 열거

sudo nmap -Pn -sV \
  --script=discovery \
  -p<PORTS> <IP> \
  -oA 03_discovery

###----------------------------------------------------------###

# safe와 discovery를 함께 실행
# 스크립트 수가 많아 오래 걸릴 수 있음

sudo nmap -Pn -sV \
  --script="safe or discovery" \
  -p<PORTS> <IP> \
  -oA 04_safe_discovery

###----------------------------------------------------------###

# default까지 포함한 넓은 범위의 정보 수집

sudo nmap -Pn -sV \
  --script="default or safe or discovery" \
  -p<PORTS> <IP> \
  -oA 05_full_enumeration

###----------------------------------------------------------###

# 알려진 취약점 후보 검사
# 출력된 CVE나 VULNERABLE 문구를 확정 취약점으로 판단하지 않음
# 대상 OS, 서비스, 버전, 설정과 일치하는지 별도 검증 필요

sudo nmap -Pn -sV \
  --script=vuln \
  -p<PORTS> <IP> \
  -oA 06_vuln_candidates

###----------------------------------------------------------###

# 인증 관련 검사

sudo nmap -Pn -sV \
  --script=auth \
  -p<PORTS> <IP> \
  -oA 07_auth

# brute, exploit, dos 카테고리 전체 실행은 기본 열거에 사용하지 않음
# 특정 서비스와 목적을 확인한 뒤 필요한 개별 스크립트만 실행

###----------------------------------------------------------###

# 특정 스크립트 하나 실행

sudo nmap -Pn -p80 \
  --script=http-enum \
  <IP>

sudo nmap -Pn -p445 \
  --script=smb-vuln-ms17-010 \
  <IP>

sudo nmap -Pn -p21 \
  --script=ftp-anon \
  <IP>

###----------------------------------------------------------###

# 여러 스크립트를 쉼표로 묶어서 실행

sudo nmap -Pn -p80,443 \
  --script="http-enum,http-title,http-server-header" \
  <IP>

# 프로토콜 와일드카드 전체 실행은 기본적으로 권장하지 않음
# 공격적이거나 불필요한 스크립트가 포함될 수 있음

sudo nmap -Pn -p80,443 --script="http-*" <IP>
sudo nmap -Pn -p139,445 --script="smb-*" <IP>

# 와일드카드가 꼭 필요하면 공격적 카테고리 제외

sudo nmap -Pn -p80,443 \
  --script="http-* and not (brute or dos or exploit or fuzzer or intrusive)" \
  <IP>

sudo nmap -Pn -p139,445 \
  --script="smb-* and not (brute or dos or exploit or fuzzer or intrusive)" \
  <IP>

###----------------------------------------------------------###

# HTTP 로그인 경로를 지정한 브루트포스
# 계정 잠금과 IP 블랙리스트 여부를 먼저 확인

sudo nmap -Pn -p80 \
  --script=http-brute \
  --script-args='http-brute.path=/admin,userdb=users.txt,passdb=pass.txt,brute.firstonly=true' \
  <IP>

###----------------------------------------------------------###

# FTP에서 첫 번째 성공 자격 증명을 찾으면 중지

sudo nmap -Pn -p21 \
  --script=ftp-brute \
  --script-args='userdb=users.txt,passdb=pass.txt,brute.firstonly=true' \
  <IP>

###----------------------------------------------------------###

# 스크립트 설명, 카테고리, 인자 확인

nmap --script-help http-enum
nmap --script-help smb-os-discovery
nmap --script-help smb-vuln-ms17-010

###----------------------------------------------------------###

# 설치된 NSE 스크립트 검색

ls /usr/share/nmap/scripts/ | grep '^http-'
ls /usr/share/nmap/scripts/ | grep '^smb-'
ls /usr/share/nmap/scripts/ | grep '^ftp-'

###----------------------------------------------------------###

# NSE 스크립트 데이터베이스 갱신

sudo nmap --script-updatedb
```

## 권장 순서

```bash
# 1. 열린 포트에 기본 NSE와 버전 탐지

sudo nmap -Pn -sC -sV \
  -p<PORTS> <IP> \
  -oA 01_default
  
###----------------------------------------------------------###

# 2. safe와 discovery로 추가 정보 수집

sudo nmap -Pn -sV \
  --script="safe or discovery" \
  -p<PORTS> <IP> \
  -oA 02_discovery

###----------------------------------------------------------###

# 3. 확인된 서비스만 서비스별 상세 스크립트 실행

sudo nmap -Pn -sV \
  --script="<SERVICE_SCRIPTS>" \
  -p<SERVICE_PORTS> <IP> \
  -oA 03_service_detail

###----------------------------------------------------------###

# 4. 실제 제품과 버전이 확인된 뒤 전용 취약점 스크립트 실행

sudo nmap -Pn -sV \
  --script=<SPECIFIC_VULN_SCRIPT> \
  -p<SERVICE_PORTS> <IP> \
  -oA 04_specific_vuln

###----------------------------------------------------------###

# 5. 넓은 vuln 카테고리는 마지막 보조 검사로만 사용

sudo nmap -Pn -sV \
  --script=vuln \
  -p<PORTS> <IP> \
  -oA 05_vuln_candidates

###----------------------------------------------------------###

# 6. 필요할 때만 개별 brute 또는 exploit 스크립트 실행
```

## 출력 형식

```bash
# 일반 텍스트 (-oN 권장)
nmap -sC -sV <IP> -oN scan.txt

# XML (자동화 파싱용)
nmap -sC -sV <IP> -oX scan.xml

# Grepable
nmap -sC -sV <IP> -oG scan.gnmap

# 모든 형식 동시
nmap -sC -sV <IP> -oA scan    # scan.nmap, scan.xml, scan.gnmap 생성

# 상세 출력
nmap -sC -sV <IP> -v
nmap -sC -sV <IP> -vv          # 더 상세
```

## 호스트 발견 (네트워크 스캔)

```bash
# Ping sweep
nmap -sn 192.168.1.0/24
nmap -sn 192.168.1.1-254

# ICMP 없이 (방화벽 우회)
nmap -Pn 192.168.1.0/24         # Ping 없이 모든 호스트 스캔
nmap -PS22,80,443 192.168.1.0/24  # SYN ping으로 호스트 발견
nmap -PA80 192.168.1.0/24       # ACK ping

# ARP ping (로컬 네트워크)
nmap -PR 192.168.1.0/24
```

## 결과 파싱

```bash
# 열린 포트만 추출
grep "open" scan.txt
grep "/tcp" scan.txt | grep "open"

# 특정 서비스 찾기
grep "http" scan.txt
grep "ssh" scan.txt

# grepable 파일에서 IP 추출
grep "Up" scan.gnmap | cut -d' ' -f2

# nmap 결과에서 포트 목록 추출
grep -oP '\d+/open' scan.gnmap | cut -d'/' -f1 | sort -n | tr '\n' ','
```
