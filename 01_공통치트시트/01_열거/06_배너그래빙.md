# 배너 그래빙 (Banner Grabbing)

서비스 버전, 소프트웨어 이름, OS 정보를 배너에서 수집하는 초기 열거 기법.

---

## 범용 도구

### netcat

```bash
# 기본 배너 수집
netcat <IP> 21
nc -nv <IP> 21
# -n : DNS 조회 없이 IP 직접 연결 (속도 향상)
# -v : verbose, 연결 상태 상세 출력

# 타임아웃 설정
nc -nv -w 3 <IP> 80
# -w 3 : 3초 동안 응답 없으면 자동 종료 (무한 대기 방지)

# 여러 포트 순차 탐색
for port in 21 22 25 80 110 143 443 3306 3389 8080; do
  echo -n "[+] Port $port: "
  nc -nv -w 2 <IP> $port 2>&1 | head -1
done
```

### nmap 배너 스크립트

```bash
# banner 스크립트 — 서비스 첫 응답 원문 캡처
nmap -p 21,22,25,80,443,3306 --script=banner <IP>
# --script=banner : 각 포트에 연결 후 서버가 먼저 보내는 배너 문자열 수집

# 서비스 버전 탐지
nmap -sV <IP>
# -sV : 서비스 버전 탐지 활성화 (프로브 패킷 보내 응답으로 버전 식별)

nmap -sV --version-intensity 9 <IP>
# --version-intensity 0~9 : 탐지 강도
#   0 = 가장 빠름, 정확도 낮음 (라이트 프로브만)
#   5 = 기본값
#   9 = 가장 느림, 정확도 높음 (모든 프로브 사용)

nmap -sV --version-all <IP>
# --version-all : 모든 프로브 시도 (--version-intensity 9 와 동일)

# 전 포트 버전 탐지
nmap -sV -p- --min-rate 5000 <IP>
# -p-          : 1~65535 전 포트 스캔
# --min-rate 5000 : 초당 최소 5000 패킷 전송 (빠르게)
```

### Python 배너 그래버

```python
#!/usr/bin/env python3
import socket

def grab_banner(ip, port, timeout=3, send=None):
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        # AF_INET  : IPv4 소켓
        # SOCK_STREAM : TCP 소켓
        s.settimeout(timeout)   # 응답 대기 최대 시간(초)
        s.connect((ip, port))
        if send:
            s.send(send)        # HTTP 등 먼저 요청 보내야 응답하는 서비스용
        banner = s.recv(1024).decode(errors='ignore').strip()
        # recv(1024) : 최대 1024바이트 수신
        # decode(errors='ignore') : 바이너리 응답 디코딩 시 오류 무시
        print(f"[{port:5d}] {banner[:100]}")
        s.close()
    except Exception as e:
        print(f"[{port:5d}] 실패: {e}")

ip = "<IP>"

# 배너를 먼저 보내는 서비스 (연결하면 자동으로 배너 전송)
passive_ports = [21, 22, 23, 25, 110, 143, 3306, 5432, 6379]
for port in passive_ports:
    grab_banner(ip, port)

# 요청을 먼저 보내야 응답하는 서비스
http_req = b"HEAD / HTTP/1.0\r\nHost: " + ip.encode() + b"\r\n\r\n"
grab_banner(ip, 80, send=http_req)
grab_banner(ip, 8080, send=http_req)
```

## FTP (21)

```bash
# netcat
nc -nv <IP> 21
# 응답 예: 220 vsftpd 2.3.4
# 응답 예: 220 ProFTPD 1.3.5 Server

# nmap
nmap -sV -p 21 <IP>
# -sV : 버전 탐지

nmap -p 21 --script=banner <IP>
# banner : 배너 원문 그대로 출력

nmap -p 21 --script=ftp-syst <IP>
# ftp-syst : SYST 명령 전송 → 서버 OS 종류 응답 (UNIX Type: L8 등)

nmap -p 21 --script=ftp-anon <IP>
# ftp-anon : 익명(anonymous) 로그인 가능 여부 확인
#            가능하면 읽기/쓰기 권한까지 표시

nmap -p 21 --script=ftp-vsftpd-backdoor <IP>
# ftp-vsftpd-backdoor : vsftpd 2.3.4 백도어 취약점 탐지
#                       취약하면 6200 포트 백도어 쉘 오픈 시도

nmap -p 21 --script=ftp-vuln-cve2010-4221 <IP>
# ftp-vuln-cve2010-4221 : ProFTPd 1.3.2rc3~1.3.3b 취약점 탐지

# 배너 해석
# 220 vsftpd 2.3.4  → CVE-2011-2523 백도어 (사용자명에 :) 입력으로 트리거)
# 220 ProFTPD 1.3.5 → CVE-2015-3306 mod_copy (인증 없이 파일 복사)
# 220 FileZilla     → 기본 자격증명 및 구버전 취약점 확인
```

## SSH (22)

```bash
# netcat — 프로토콜 버전 + 소프트웨어 버전 확인
nc -nv <IP> 22
# 응답 예: SSH-2.0-OpenSSH_7.4p1 Debian-10+deb9u7
# 구조: SSH-<프로토콜버전>-<소프트웨어명>_<버전> <OS정보>

# nmap
nmap -sV -p 22 <IP>

nmap -p 22 --script=ssh-hostkey <IP>
# ssh-hostkey : 서버의 호스트 공개키 지문 출력
#               동일 키가 다른 호스트에서도 발견되면 키 재사용 의심

nmap -p 22 --script=ssh2-enum-algos <IP>
# ssh2-enum-algos : 서버가 지원하는 암호화 알고리즘 목록
#   - kex (키 교환), encryption (암호화), mac (무결성), compression (압축)
#   구형 알고리즘(arcfour, 3des 등) 사용 시 취약점 가능성

nmap -p 22 --script=ssh-auth-methods \
  --script-args="ssh.user=root" <IP>
# ssh-auth-methods : 특정 사용자에게 허용된 인증 방식 목록
#   password   = 패스워드 인증 허용 → 브루트포스 가능
#   publickey  = 키 인증만 허용
#   none       = 인증 없이 접근 가능 (매우 취약)
# --script-args="ssh.user=root" : 확인할 사용자명 지정

# 배너 해석
# SSH-2.0-OpenSSH_7.4p1 → CVE-2018-15473 (7.7 미만: 사용자 열거 가능)
# SSH-2.0-OpenSSH_4.x   → 매우 구버전, searchsploit으로 전수 확인
# Debian/Ubuntu 정보 포함 시 → OS 버전 추측 가능
```

## SMTP (25 / 587 / 465)

```bash
# netcat
nc -nv <IP> 25
# 응답 예: 220 mail.corp.com ESMTP Postfix (Ubuntu)
# 응답 예: 220 mail.corp.com ESMTP Sendmail 8.14.4

# EHLO 명령으로 지원 기능 확인
nc -nv <IP> 25
EHLO test.com
# EHLO : 서버에 클라이언트를 소개하고 지원 기능 목록 요청
# 응답 항목 의미:
# 250-PIPELINING     : 여러 명령 한 번에 전송 가능
# 250-SIZE 10240000  : 최대 메일 크기
# 250-VRFY           : 사용자 존재 확인 명령 허용 → 사용자 열거 가능
# 250-EXPN           : 메일링 리스트 확장 허용
# 250-AUTH LOGIN PLAIN CRAM-MD5 : 지원 인증 방식

# nmap
nmap -sV -p 25,587,465 <IP>

nmap -p 25 --script=smtp-commands <IP>
# smtp-commands : EHLO 전송 후 서버가 지원하는 명령어 전체 목록
#                 VRFY, EXPN 허용 여부 확인 가능

nmap -p 25 --script=smtp-ntlm-info \
  --script-args="smtp-ntlm-info.domain=corp.local" <IP>
# smtp-ntlm-info : NTLM 인증 시도 시 서버가 내부 정보 노출
#   - 내부 호스트명, NetBIOS 이름, DNS 도메인, OS 버전 획득 가능
# --script-args="smtp-ntlm-info.domain" : 도메인 지정

nmap -p 25 --script=smtp-open-relay \
  --script-args="smtp-open-relay.domain=corp.local" <IP>
# smtp-open-relay : 외부 도메인으로 릴레이 허용 여부 테스트
#                   허용 시 스팸/피싱 전송 가능

nmap -p 25 --script=smtp-enum-users \
  --script-args="smtp-enum-users.methods=VRFY,RCPT,EXPN" <IP>
# smtp-enum-users : VRFY/RCPT/EXPN 방식으로 사용자 열거
# --script-args="smtp-enum-users.methods" : 사용할 방식 지정
#   VRFY  = VRFY <user> 명령으로 확인 (직접적)
#   RCPT  = MAIL FROM + RCPT TO 방식 (VRFY 막혔을 때)
#   EXPN  = 메일링 리스트 확장으로 확인
```

## HTTP / HTTPS (80 / 443 / 8080 / 8443)

```bash
# curl
curl -I http://<IP>
# -I : HEAD 요청 (본문 없이 헤더만 수신, 빠름)

curl -I https://<IP> -k
# -k : --insecure, SSL/TLS 인증서 오류 무시

curl -sv http://<IP> 2>&1 | grep "^< "
# -s : --silent, 진행률 표시 제거
# -v : --verbose, 요청/응답 헤더 전체 출력
# 2>&1 : stderr(헤더)를 stdout으로 합치기
# grep "^< " : 서버 응답 헤더만 추출 (< 로 시작하는 줄)

# 핵심 헤더 추출
curl -s -I http://<IP> | grep -i \
  "server\|x-powered-by\|x-generator\|x-aspnet\|via\|x-backend\|x-cms"
# Server          : 웹서버 종류/버전 (Apache/2.4.49, nginx/1.14.0 등)
# X-Powered-By    : 백엔드 언어 버전 (PHP/5.6.40, ASP.NET 4.0 등)
# X-Generator     : CMS 정보 (WordPress 5.8, Drupal 9 등)
# X-AspNet-Version: ASP.NET 버전
# Via             : 프록시/로드밸런서 정보
# X-Backend       : 백엔드 서버 정보 (내부 IP 노출 가능)

# whatweb
whatweb http://<IP>
# 기본 탐지: 웹서버, CMS, 프레임워크, 언어, 쿠키 등

whatweb -v http://<IP>
# -v : verbose, 각 플러그인이 탐지한 상세 정보 출력
#      버전, 증거(어느 헤더/본문에서 발견) 포함

whatweb -a 1 http://<IP>
# -a 1 : --aggression 1, 스텔스 (기본값, 요청 최소화)

whatweb -a 3 http://<IP>
# -a 3 : 공격적 탐지, 더 많은 URL 크롤링 및 프로브 전송
#   1 = Stealthy (기본, 1요청)
#   2 = Unused
#   3 = Aggressive (더 많은 요청, 더 많은 정보)
#   4 = Heavy (모든 플러그인, 느림)

whatweb -a 3 http://<IP> --log-verbose=whatweb.txt
# --log-verbose : 상세 결과를 파일로 저장

# nmap HTTP 스크립트
nmap -p 80,443,8080,8443 --script=http-server-header <IP>
# http-server-header : HTTP Server 헤더 값만 추출

nmap -p 80 --script=http-headers <IP>
# http-headers : 전체 HTTP 응답 헤더 출력

nmap -p 80 --script=http-title <IP>
# http-title : 페이지 <title> 태그 내용 출력 (사이트 목적 파악)

nmap -p 80 --script=http-methods \
  --script-args="http-methods.url-path=/admin" <IP>
# http-methods : 서버가 허용하는 HTTP 메서드 확인
#   GET, POST는 기본. PUT, DELETE, TRACE는 위험 가능성
# --script-args="http-methods.url-path" : 확인할 경로 지정

nmap -p 80 --script=http-auth-finder <IP>
# http-auth-finder : 인증 방식 탐지
#   Basic, Digest, NTLM, Form-based 등 구분

nmap -p 80 --script=http-php-version <IP>
# http-php-version : PHP 버전 탐지
#   /?=PHPE9568F34-D428-11d2-A769-00AA001ACF42 같은 특수 요청으로 확인

nmap -p 80 --script=http-default-accounts \
  --script-args="http-default-accounts.fingerprintfile=http-default-accounts-fingerprints.lua" <IP>
# http-default-accounts : 알려진 기본 자격증명 자동 시도
#   Tomcat(admin/admin), Jenkins, 라우터 관리페이지 등

nmap -p 80 --script=http-git <IP>
# http-git : /.git/ 디렉터리 노출 여부 확인
#            노출 시 소스코드, 자격증명, 히스토리 다운로드 가능

nmap -p 80 --script=http-robots.txt <IP>
# http-robots.txt : robots.txt 내용 출력
#                   숨겨진 관리자 경로, 민감한 디렉터리 발견 가능
```

## SMB (139 / 445)

```bash
# nmap
nmap -sV -p 139,445 <IP>

nmap -p 445 --script=smb-os-discovery <IP>
# smb-os-discovery : SMB 프로토콜 통해 OS 정보 수집
#   - OS 버전 (Windows 10, Server 2019 등)
#   - 컴퓨터 이름, 도메인/워크그룹, 빌드 번호

nmap -p 445 --script=smb-security-mode <IP>
# smb-security-mode : SMB 보안 설정 확인
#   - 인증 방식 (user-level / share-level)
#   - Message Signing : enabled/disabled
#     disabled → SMB Relay 공격 가능

nmap -p 445 --script=smb-protocols <IP>
# smb-protocols : 서버가 지원하는 SMB 버전 목록
#   SMBv1 지원 확인 → EternalBlue (MS17-010) 가능성 높음

nmap -p 445 --script=smb2-security-mode <IP>
# smb2-security-mode : SMB2/3 서명 설정 확인
#   "Message signing enabled but not required" → Relay 가능

nmap -p 445 --script=smb-vuln-ms17-010 <IP>
# smb-vuln-ms17-010 : EternalBlue 취약점 직접 탐지
#   VULNERABLE → Metasploit ms17_010_eternalblue 사용 가능

nmap -p 445 --script=smb-vuln-ms08-067 <IP>
# smb-vuln-ms08-067 : MS08-067 취약점 탐지
#   Windows XP/2003 대상 RCE

nmap -p 139,445 --script="smb-vuln*" <IP>
# smb-vuln* : smb-vuln 접두사 스크립트 전체 실행
#             MS06-025, MS07-029, MS08-067, MS10-054, MS17-010 등

# netexec
nxc smb <IP>
# 출력: SMB <IP> 445 HOSTNAME [*] Windows 10.0 Build 19041 x64 (domain:CORP) (signing:False) (SMBv1:False)
# 한 번에 OS버전, 빌드, 도메인, SMB 서명 여부, SMBv1 활성화 확인
```

## MySQL (3306)

```bash
# netcat — 연결 즉시 MySQL 핸드셰이크 배너 수신
nc -nv <IP> 3306
# 응답: 바이너리지만 버전 문자열 포함
# 예: ...5.7.32-log... 또는 ...8.0.21...

# nmap
nmap -sV -p 3306 <IP>

nmap -p 3306 --script=mysql-info <IP>
# mysql-info : MySQL 버전, 프로토콜 버전, 기능 플래그 출력
#   - Version: 5.7.32
#   - Protocol: 10
#   - Capabilities: 지원 기능 목록

nmap -p 3306 --script=mysql-empty-password <IP>
# mysql-empty-password : root 및 anonymous 계정 빈 패스워드 여부 확인
#                        접근 가능하면 즉시 exploit 가능

nmap -p 3306 --script=mysql-databases \
  --script-args="mysqluser=root,mysqlpass=''" <IP>
# mysql-databases : 접근 가능한 DB 목록 출력
# --script-args : 접속에 사용할 사용자명/패스워드 지정

nmap -p 3306 --script=mysql-users \
  --script-args="mysqluser=root,mysqlpass=''" <IP>
# mysql-users : 등록된 MySQL 사용자 목록 출력

nmap -p 3306 --script=mysql-brute \
  --script-args="userdb=users.txt,passdb=pass.txt" <IP>
# mysql-brute : 사용자명/패스워드 브루트포스 시도
```

## RDP (3389)

```bash
# nmap
nmap -sV -p 3389 <IP>

nmap -p 3389 --script=rdp-enum-encryption <IP>
# rdp-enum-encryption : RDP 암호화 수준 및 프로토콜 확인
#   - CredSSP (NLA) : 자격증명 먼저 필요 → 브루트포스 더 어려움
#   - TLS            : 인증서 기반
#   - Classic RDP    : 구형, NLA 없음 → 로그인 화면 접근 가능

nmap -p 3389 --script=rdp-vuln-ms12-020 <IP>
# rdp-vuln-ms12-020 : CVE-2012-0002 DoS 취약점 탐지
#                     (BlueKeep과 다른 취약점)

# netexec
nxc rdp <IP>
# 한 번에 OS버전, 빌드번호, 호스트명, NLA 여부, 도메인 확인
```

## SNMP (161 UDP)

```bash
# nmap (UDP 스캔 필수)
nmap -sU -sV -p 161 <IP>
# -sU : UDP 스캔 (SNMP는 UDP 161 사용)

nmap -sU -p 161 --script=snmp-info <IP>
# snmp-info : community string "public" 으로 기본 시스템 정보 수집
#   - sysDescr : OS 및 버전 문자열
#   - sysUpTime : 부팅 후 경과 시간
#   - sysContact : 관리자 연락처
#   - sysLocation : 장치 위치

nmap -sU -p 161 --script=snmp-sysdescr \
  --script-args="snmpcommunity=public" <IP>
# snmp-sysdescr : sysDescr OID 값만 출력 (OS 버전 핵심 정보)
# --script-args="snmpcommunity" : 사용할 community string 지정

nmap -sU -p 161 --script=snmp-brute \
  --script-args="snmp-brute.communitiesdb=/usr/share/seclists/Discovery/SNMP/common-snmp-community-strings.txt" <IP>
# snmp-brute : community string 브루트포스
# --script-args="snmp-brute.communitiesdb" : 사용할 워드리스트 경로

# snmpwalk (community string 알고 있을 때)
snmpwalk -v 2c -c public <IP> 1.3.6.1.2.1.1.1.0
# -v 2c : SNMP 버전 2c (1 또는 2c 주로 사용, 3은 인증 필요)
# -c public : community string (읽기용 기본값)
# 1.3.6.1.2.1.1.1.0 : sysDescr OID (OS 정보)

snmpwalk -v 2c -c public <IP> 1.3.6.1.2.1.1.5.0
# 1.3.6.1.2.1.1.5.0 : sysName OID (호스트명)
```

## POP3 (110) / IMAP (143)

```bash
# netcat
nc -nv <IP> 110    # POP3

nmap -p 110 --script=pop3-brute \
```

## 팁

- **표준 형식이 아닌 이상한 문자열이 나오면 그건 우연이 아니라 신호다** (예: DNS 배너가 `9.16.1-Ubuntu` 같은 정상 패턴이 아니라 `1337_HTB_DNS`처럼 커스텀 티가 나는 값일 때) — 관리자가 의도적으로 숨김/변조했거나, CTF/HTB식 이스터에그 힌트일 가능성. 문자열 자체를 검색해보고, 해당 서비스에 zone transfer 등 추가 시도로 이어갈 것
- DNS 배너는 `dig version.bind CHAOS TXT @<IP>` 로 확인 (자세한 건 `08_DNS_열거` 참고)
- 배너를 숨기거나 조작하는 서버도 있어서 다중 확인 필요
- HTTP에서 버전 없이 `Server: Apache` 만 나오면 에러 페이지(404/500)에서 버전 노출되는 경우 있음
- `X-Powered-By`, `X-Generator`, `X-AspNet-Version` 등 커스텀 헤더도 반드시 확인
- SMB `smb-os-discovery` 로 OS 버전 확인하면 RDP, SSH 취약점도 동시에 매핑 가능
- SNMP는 UDP라 `sU` 빠뜨리면 결과 안 나옴
- 수집한 버전은 searchsploit + `site:exploit-db.com <버전>` Google 검색 병행
